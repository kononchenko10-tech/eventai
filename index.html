document.getElementById("genBtn").onclick = function() { var holiday = document.getElementById("holiday").value; var classFrom = parseInt(document.getElementById("classFrom").value); var resultDiv = document.getElementById("result"); resultDiv.innerText = "Собираю новый сценарий..."; var csvUrl = "https://docs.google.com/spreadsheets/d/1c0uE0lfqJLFeHzI8g1oO1qhMa5MZrwesTTEvJLL1X8/export?format=tsv&gid=0"; fetch(csvUrl) .then(function(r) { return r.text(); }) .then(function(csvText) { var rows = csvText.split("\n"); var scripts = []; // 1. Ищем все подходящие сценарии for (var i = 1; i < rows.length; i++) { if (!rows[i].trim()) continue; var cols = rows[i].split(","); if (cols[0] && cols[0].includes(holiday)) { var dbNum = parseInt(cols[1] || ""); if (!isNaN(dbNum) && dbNum >= classFrom && dbNum <= classFrom + 3) { if (cols[7] && cols[7].length > 50) { scripts.push(cols[7].replace(/;/g, ",")); } } } } if (scripts.length < 2) { resultDiv.innerText = "Для сборки нужно минимум 2 сценария. Добавьте больше данных в таблицу."; return; } // 2. Разбираем на части и перемешиваем var allParagraphs = []; scripts.forEach(function(text) { // Делим на абзацы var parts = text.split(/\n\n|\n/); parts.forEach(function(p) { if (p.trim().length > 30) allParagraphs.push(p.trim()); }); }); // Перемешиваем абзацы allParagraphs.sort(function() { return 0.5 - Math.random(); }); // 3. Собираем новый сценарий var finalText = "✨ Сгенерирован новый уникальный сценарий:\n\n"; finalText += "--- ВСТУПЛЕНИЕ ---\n"; finalText += allParagraphs[0] + "\n\n"; finalText += "А теперь давайте поиграем и посоревнуемся!\n\n"; finalText += "--- КОНКУРСЫ ---\n"; var contestCount = Math.min(3, allParagraphs.length - 2); for (var i = 1; i <= contestCount; i++) { finalText += "• " + allParagraphs[i] + "\n\n"; } finalText += "--- ФИНАЛ ---\n"; finalText += "Наш праздник подходит к концу. Спасибо всем за участие!\n"; finalText += allParagraphs[allParagraphs.length - 1]; resultDiv.innerText = finalText; }) .catch(function(e) { resultDiv.innerText = "Ошибка загрузки базы: " + e.message; }); };
